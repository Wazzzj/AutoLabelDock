# 按标注框裁剪图片（内置工具）
from autolabel import project, label_io

proj = project.open(".")  # 工作目录 = 项目根目录
version = "v1-产线A"  # 由右侧面板选择注入
pad = 0.08  # 外扩比例

for img in proj.list_images(data_folder=version):
    ia = label_io.load_annotation(proj.label_path_for(img))
    for ann in ia.confirmed_annotations():
        box = ann.pixel_bbox(ia.image_size, pad=pad)
        img.crop_save(box, out_dir="crops")
